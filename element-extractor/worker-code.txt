function object(v) {
  return v !== null && typeof v === "object" && !Array.isArray(v) && (Object.getPrototypeOf(v) === Object.prototype || Object.getPrototypeOf(v) === null);
}
function keys(v, allowed) {
  return object(v) && Object.keys(v).every((k) => allowed.includes(k));
}
function bounded(s, max) {
  return typeof s === "string" && s.length > 0 && s.length <= max;
}
function canonical(value, depth = 0) {
  check(depth <= 16, "json_depth", "JSON nesting exceeds the limit.");
  if (value === null || typeof value === "string" || typeof value === "boolean")
    return JSON.stringify(value);
  if (typeof value === "number") {
    check(
      Number.isFinite(value),
      "nonfinite_number",
      "Numbers must be finite."
    );
    return JSON.stringify(value);
  }
  if (Array.isArray(value))
    return "[" + Array.from(value, (v) => canonical(v, depth + 1)).join(",") + "]";
  check(object(value), "invalid_json", "Only plain JSON values are supported.");
  return "{" + Object.keys(value).sort().map((k) => JSON.stringify(k) + ":" + canonical(value[k], depth + 1)).join(",") + "}";
}
var copy = (v) => JSON.parse(canonical(v));
async function digest(value) {
  const bytes = await crypto.subtle.digest(
    "SHA-256",
    encoder.encode(typeof value === "string" ? value : canonical(value))
  );
  return Array.from(
    new Uint8Array(bytes),
    (n) => n.toString(16).padStart(2, "0")
  ).join("");
}
function freeze(v) {
  if (v && typeof v === "object") {
    Object.values(v).forEach(freeze);
    Object.freeze(v);
  }
  return v;
}

async function createPacket({ source, schema }, now = (/* @__PURE__ */ new Date()).toISOString()) {
  check(
    keys(source, ["url", "text"]) && bounded(source.text, LIMITS.text),
    "invalid_source",
    "Provide nonempty source text up to 16,000 UTF-16 code units. Text is never silently truncated."
  );
  let url = null;
  if (source.url !== void 0 && source.url !== null) {
    let u;
    try {
      u = new URL(source.url);
    } catch {
    }
    check(
      u?.protocol === "https:" && !u.username && !u.password && source.url.length <= 2048,
      "invalid_source",
      "Source attribution must be a credential-free HTTPS URL. No URL is fetched in this offline lane."
    );
    url = u.href;
  }
  check(
    typeof now === "string" && Number.isFinite(Date.parse(now)),
    "invalid_timestamp",
    "Invalid receipt timestamp."
  );
  const text = source.text.replace(/\r\n?/g, "\n");
  const passages = [];
  let start = 0;
  const add = (end) => {
    const slice = text.slice(start, end);
    const leading = slice.length - slice.trimStart().length;
    const value = slice.trim();
    if (value)
      passages.push({
        id: "p" + String(passages.length + 1).padStart(3, "0"),
        text: value,
        start: start + leading,
        end: start + leading + value.length
      });
  };
  for (const m of text.matchAll(/\n[ \t]*\n+/g)) {
    add(m.index);
    start = m.index + m[0].length;
  }
  add(text.length);
  check(
    passages.length > 0 && passages.length <= LIMITS.passages,
    "passage_limit",
    "Source must contain 1\u201332 paragraphs. No paragraphs are silently omitted."
  );
  for (const p of passages) p.hash = await digest(p.text);
  const packet = {
    version: PACKET_VERSION,
    source: {
      url,
      provenance: "user_supplied_text",
      received_at: new Date(now).toISOString(),
      text,
      text_hash: await digest(text),
      original_text_hash: await digest(source.text),
      offset_unit: "UTF-16 code units; start inclusive, end exclusive"
    },
    schema: schemaOf(schema),
    passages
  };
  return freeze({ ...packet, hash: await digest(packet) });
}
async function verifyPacket(packet) {
  check(
    keys(packet, ["version", "source", "schema", "passages", "hash"]) && packet.version === PACKET_VERSION,
    "invalid_packet",
    "Unsupported packet format."
  );
  const { hash, ...body } = packet;
  check(
    hash === await digest(body),
    "packet_hash_mismatch",
    "Packet content does not match its hash."
  );
  check(
    keys(packet.source, [
      "url",
      "provenance",
      "received_at",
      "text",
      "text_hash",
      "original_text_hash",
      "offset_unit"
    ]) && packet.source.provenance === "user_supplied_text",
    "invalid_packet",
    "Unsupported source provenance."
  );
  const rebuilt = await createPacket(
    {
      source: { url: packet.source.url, text: packet.source.text },
      schema: packet.schema
    },
    packet.source.received_at
  );
  check(
    packet.source.text_hash === rebuilt.source.text_hash && canonical(packet.passages) === canonical(rebuilt.passages) && canonical(packet.schema) === canonical(rebuilt.schema),
    "invalid_packet",
    "Passage offsets, hashes or schema are inconsistent with the retained snapshot."
  );
  check(
    /^[a-f0-9]{64}$/.test(packet.source.original_text_hash),
    "invalid_packet",
    "Invalid original text hash."
  );
  return packet;
}
var issue = (code, message) => ({ code, message });
function evidenceFor(candidate, field, packet) {
  const issues = [], evidence = [];
  if (!object(candidate)) {
    if (field.required)
      issues.push(
        issue(
          "missing_field",
          "No candidate supplied for this required field."
        )
      );
    return { value: null, evidence, issues };
  }
  const value = Object.hasOwn(candidate, "value") ? candidate.value : null;
  if (!keys(candidate, ["value", "evidence"]))
    issues.push(issue("unknown_field_metadata", "Unexpected field metadata."));
  if (value === null) {
    if (field.required)
      issues.push(
        issue("required_value_missing", "A required value is missing.")
      );
    if (candidate.evidence?.length)
      issues.push(
        issue("evidence_without_value", "Evidence supplied without a value.")
      );
    return { value, evidence, issues };
  }
  const correctType = typeof value === field.type && (field.type !== "number" || Number.isFinite(value));
  if (!correctType)
    issues.push(
      issue(
        "wrong_type",
        "Candidate type does not match the requested column."
      )
    );
  if (field.type === "string" && !bounded(value, 2048))
    issues.push(
      issue("invalid_string", "Text values must contain 1\u20132048 code units.")
    );
  if (correctType && field.type === "number" && (field.min !== void 0 && value < field.min || field.max !== void 0 && value > field.max))
    issues.push(
      issue("out_of_bounds", "Number is outside the requested bounds.")
    );
  if (!Array.isArray(candidate.evidence) || candidate.evidence.length === 0 || candidate.evidence.length > LIMITS.spans) {
    issues.push(
      issue(
        "evidence_required",
        "Supply 1\u20134 supporting spans per nonempty field."
      )
    );
    return { value, evidence, issues };
  }
  for (const span of candidate.evidence) {
    if (!keys(span, ["passage_id", "quote", "occurrence"]) || !bounded(span.quote, LIMITS.quote)) {
      issues.push(issue("invalid_evidence", "Malformed supporting span."));
      continue;
    }
    const p = packet.passages.find((p2) => p2.id === span.passage_id);
    if (!p) {
      issues.push(
        issue("unknown_passage", "Evidence references an unknown passage.")
      );
      continue;
    }
    const positions = [];
    let at = 0;
    while (at <= p.text.length) {
      const found = p.text.indexOf(span.quote, at);
      if (found < 0) break;
      positions.push(found);
      at = found + 1;
    }
    if (!positions.length) {
      issues.push(
        issue(
          "quote_not_found",
          "Exact quotation is absent from the cited passage."
        )
      );
      continue;
    }
    const occurrence = span.occurrence ?? (positions.length === 1 ? 0 : null);
    if (!Number.isInteger(occurrence) || occurrence < 0 || occurrence >= positions.length) {
      issues.push(
        issue(
          "ambiguous_quote",
          "Repeated quotations require a valid zero-based occurrence index."
        )
      );
      continue;
    }
    const local = positions[occurrence];
    evidence.push({
      passage_id: p.id,
      quote: span.quote,
      occurrence,
      passage_start: local,
      passage_end: local + span.quote.length,
      source_start: p.start + local,
      source_end: p.start + local + span.quote.length,
      source_hash: packet.source.text_hash
    });
  }
  if (correctType && evidence.length) {
    const supported = field.type === "string" ? evidence.some((e) => e.quote.includes(value)) : evidence.some(
      (e) => Array.from(
        e.quote.matchAll(
          /(?<![\w.,])[-+]?(?:\d+(?:\.\d+)?|\.\d+)(?!\w|\.\d|,\d)/g
        ),
        (m) => Number(m[0])
      ).includes(value)
    );
    if (!supported)
      issues.push(
        issue(
          "literal_support_missing",
          "The literal value is not supported by the retained quotations. Numeric formatting is limited to signed, ungrouped decimal notation."
        )
      );
  }
  return { value, evidence, issues };
}

async function runInterpretation(packet, interpreter = null, cache = /* @__PURE__ */ new Map()) {

      format: "response-cache-v1",

      packet_hash: packet.hash,
      interpreter: copy(interpreter.identity),

      raw_hash: await digest(raw)

      status: "validated",

      cache_entry: freeze(copy(entry)),

      status: "invalid_response",
      packet,
      cache_entry: freeze(copy(entry)),
      cache_hit,
      validation: null,
      issues: [issue(error.code, error.message)]

async function replayReviews(packet, validation, events = []) {

    validation.packet_hash === packet.hash,

    "Validation does not belong to this packet."

    Array.isArray(events) && events.length <= LIMITS.events,
    "review_limit",
    "Review event limit exceeded."
  );

    check(
      keys(event, [
        "sequence",
        "previous_hash",
        "packet_hash",
        "validation_hash",
        "record_id",
        "decision",
        "reviewer",
        "note",
        "confirmed_association",
        "at",
        "hash"
      ]) && event.sequence === index + 1 && event.previous_hash === previous && event.packet_hash === packet.hash && event.validation_hash === validation_hash && states.has(event.record_id),
      "review_identity",

    );
    const { hash, ...body } = event;

      hash === await digest(body),
      "review_integrity",
      "Review event hash mismatch."
    );
    check(
      ["approved", "rejected", "needs_review"].includes(event.decision) && bounded(event.reviewer, 80) && event.reviewer.trim().length > 0 && bounded(event.note, 1e3) && event.note.trim().length > 0 && typeof event.confirmed_association === "boolean" && typeof event.at === "string" && Number.isFinite(Date.parse(event.at)),
      "invalid_review",
      "Malformed review event."
    );
    if (event.decision === "approved")
      check(
        !validation.issues.length && !blocking(validation.records.find((r) => r.id === event.record_id)) && event.confirmed_association,
        "approval_blocked",
        "Approval requires clean independent checks and an explicit human association confirmation."
      );
    states.set(event.record_id, event.decision);
    previous = hash;
  }
  return states;
}

  const body = {
    ...copy(decision),
    confirmed_association: decision.confirmed_association ?? false,
    sequence: events.length + 1,
    previous_hash: events.at(-1)?.hash ?? null,
    packet_hash: packet.hash,
    validation_hash: await digest(validation),
    at: now
  };
  const updated = [...copy(events), { ...body, hash: await digest(body) }];
  await replayReviews(packet, validation, updated);
  return freeze(updated);
}
function quoteCSV(value) {
  let text = String(value ?? "");
  if (/^[\s\u0000-\u001f]*[=+@-]/.test(text)) text = "'" + text;
  return '"' + text.replace(/"/g, '""') + '"';
}
async function exportInterpretation(packet, entry, events = [], { approvedOnly = true } = {}) {
  const validation = await revalidate(packet, entry), states = await replayReviews(packet, validation, events);
  const all = validation.records.map((r) => ({
    ...copy(r),
    review_status: states.get(r.id)
  }));
  const records = approvedOnly ? all.filter((r) => r.review_status === "approved") : all;
  const json2 = {
    kind: approvedOnly ? "approved_interpretation_export" : "interpretation_diagnostic_export",
    counts: {
      total_candidates: all.length,
      exported: records.length,
      excluded: all.length - records.length
    },
    notice: "Scripted candidate data; not AI inference or automatic factual verification. Only explicitly reviewed records may be approved.",
    packet: copy(packet),
    cache_identity: {
      key: entry.key,
      raw_hash: entry.raw_hash,
      interpreter: copy(entry.interpreter)
    },
    validation: { ...copy(validation), records },
    review_events: copy(events)
  };
  if (!approvedOnly) json2.raw_response = copy(entry.raw_response);
  const headers2 = [
    "record_id",
    "review_status",
    ...packet.schema.fields.map((f) => f.name),
    "validation_issues"
  ];
  const csv = [
    headers2,
    ...records.map((r) => [
      r.id,
      r.review_status,
      ...packet.schema.fields.map((f) => r.fields[f.name].value),
      [
        ...r.issues.map((i) => i.code),
        ...Object.entries(r.fields).flatMap(
          ([name, f]) => f.issues.map((i) => name + ":" + i.code)
        ),
        ...validation.issues.map((i) => i.code)
      ].join(";")
    ])
  ].map((row) => row.map(quoteCSV).join(",")).join("\r\n") + "\r\n";
  return { json: json2, csv };
}

        if (bytes > 131072) {
          await reader.cancel();
          fail("request_limit", "The lab request exceeds 128 KiB.", 413);
        }
        chunks.push(value);
      }
    } finally {
      reader.releaseLock();
    }
  let body;
  try {
    body = JSON.parse(await new Blob(chunks).text());
  } catch {
    fail("invalid_json", "The lab request is not valid JSON.");
  }
  return body;
}
async function checkJob(job) {
  if (!only(job, ["example_id", "packet", "cache_entry", "events"]) || !Object.hasOwn(examples, job.example_id) || !Array.isArray(job.events) || job.events.length > 64)
    fail(
      "invalid_lab_job",
      "Invalid lab session. Only the built-in fictional examples and up to 64 review events are supported."
    );
  const fixture = examples[job.example_id];
  if (job.packet?.source?.text !== fixture.source.text || job.packet?.source?.url !== fixture.source.url || canonical(job.packet?.schema) !== canonical(fixture.schema) || canonical(job.cache_entry?.raw_response) !== canonical(fixture.scripted_response))
    fail(
      "example_mismatch",
      "The lab accepts only its unchanged fictional source, columns and scripted response. Start the example again."
    );
  await revalidate(job.packet, job.cache_entry);
  return exportInterpretation(job.packet, job.cache_entry, job.events, {
    approvedOnly: false
  });
}
function view(job, result) {
  const records = result.json.validation.records;
  return {
    job,
    validation: result.json.validation,
    counts: {
      candidates: records.length,
      approved: records.filter((r) => r.review_status === "approved").length,
      rejected: records.filter((r) => r.review_status === "rejected").length,
      pending: records.filter((r) => r.review_status === "needs_review").length
    },
    notice: "Fictional scripted examples. No AI inference. Session-only, self-declared review; not an authenticated audit service."
  };
}
async function labFetch(request) {
  const url = new URL(request.url);
  if (url.pathname === "/interpretation" || url.pathname === "/interpretation-client.js") {
    if (request.method !== "GET")
      return new Response("Method not allowed", {
        status: 405,
        headers: { ...headers, Allow: "GET" }
      });
    return new Response(
      url.pathname === "/interpretation" ? LAB_PAGE : lab_client_default,
      {
        headers: {
          ...headers,
          "Content-Type": url.pathname === "/interpretation" ? "text/html; charset=utf-8" : "application/javascript; charset=utf-8"
        }
      }
    );
  }
  if (request.method !== "POST") {
    const response = reply(
      {
        error: "Use a same-origin JSON POST request.",
        code: "method_not_allowed"
      },
      405
    );
    response.headers.set("Allow", "POST");
    return response;
  }
  const origin = request.headers.get("origin");
  if (origin && origin !== url.origin)
    return reply(
      {
        error: "Cross-origin lab requests are not supported.",
        code: "origin_not_allowed"
      },
      403
    );
  try {
    const body = await readBody(request);
    if (url.pathname === "/api/interpretation/start") {
      if (!only(body, ["example_id"]) || !Object.hasOwn(examples, body.example_id))
        fail(
          "unknown_example",
          "Choose one of the built-in fictional examples."
        );
      const fixture = examples[body.example_id], packet = await createPacket(fixture);
      const run = await runInterpretation(
        packet,
        await scriptedInterpreter(fixture.scripted_response)
      );
      const job = {
        example_id: body.example_id,
        packet,
        cache_entry: run.cache_entry,
        events: []
      };
      return reply(view(job, await checkJob(job)));
    }
    if (url.pathname !== "/api/interpretation/action")
      return reply({ error: "Not found", code: "not_found" }, 404);
    if (!only(body, ["action", "job", "decision", "format", "diagnostic"]) || !["revalidate", "review", "export"].includes(body.action))
      fail("invalid_action", "Choose revalidate, review or export.");
    const result = await checkJob(body.job);
    if (body.action === "review") {
      if (body.job.events.length >= 64)
        fail(
          "review_limit",
          "This demo is limited to 64 review events. Download a diagnostic report, then load a new example.",
          413
        );
      const events = await appendReview(
        body.job.packet,
        body.job.cache_entry,
        body.job.events,
        body.decision
      );
      const job = { ...body.job, events };
      return reply(view(job, await checkJob(job)));
    }
    if (body.action === "revalidate") return reply(view(body.job, result));
    if (!["csv", "json"].includes(body.format) || body.diagnostic !== void 0 && typeof body.diagnostic !== "boolean")
      fail(
        "invalid_export",
        "Choose CSV or JSON and a boolean diagnostic flag."
      );
    const exported = await exportInterpretation(
      body.job.packet,
      body.job.cache_entry,
      body.job.events,
      { approvedOnly: !body.diagnostic }
    );
    const prefix = body.diagnostic ? "interpretation-diagnostic" : "interpretation-approved";
    return new Response(
      body.format === "csv" ? exported.csv : JSON.stringify(exported.json, null, 2) + "\n",
      {
        headers: {
          ...headers,
          "Content-Type": body.format === "csv" ? "text/csv; charset=utf-8" : "application/json; charset=utf-8",
          "Content-Disposition": `attachment; filename="${prefix}.${body.format}"`,
          "X-Exported-Records": String(exported.json.counts.exported)
        }
      }
    );
  } catch (error) {
    return reply(
      {
        error: error instanceof ContractError ? error.message : "The lab request failed. Your current session has not been replaced.",
        code: error instanceof ContractError ? error.code : "lab_request_failed"
      },
      error.status || (error instanceof ContractError ? 400 : 500)
    );
  }
}

function isInvalidCodePoint(codePoint) {
  return codePoint === 0 || codePoint >= 55296 && codePoint <= 57343 || codePoint > 1114111;

function replaceCodePoint(codePoint) {
  if (isInvalidCodePoint(codePoint)) {
    return 65533;
  }
  if (codePoint >= 128 && codePoint <= 159) {
    return c1[codePoint - 128] || codePoint;
  }
  return codePoint;

function codePointToString(codePoint) {

function decodeTrieDict(input, resultLength, atomCount, dict1AtomCount, ngramCount, dictSize) {
  const base = 91;
  const inputLength = input.length;
  const twoCharBias = dictSize * (base - 1);
  let pos = 0;
  const readSlotCode = () => {
    const c12 = BASE91_INVERSE[input.charCodeAt(pos++)];
    return c12 < dictSize ? c12 : c12 * base - twoCharBias + BASE91_INVERSE[input.charCodeAt(pos++)];
  };
  const dict2AtomCount = atomCount - dict1AtomCount;
  const slotCount = atomCount + ngramCount;
  const single = new Int32Array(slotCount);
  single.fill(-1, dict1AtomCount, dictSize);
  single.fill(-1, dictSize + dict2AtomCount, slotCount);
  const start = new Int32Array(slotCount);
  const length = new Int32Array(slotCount);
  function decodeDelta(count, off) {
    let previous = 0;
    let slot = off;
    const end = off + count;
    while (slot < end) {
      const code = BASE91_INVERSE[input.charCodeAt(pos++)];
      if (code < 89) {
        previous += code;
        single[slot++] = previous;
      } else if (code === 89) {
        let runLength = BASE91_INVERSE[input.charCodeAt(pos++)] + 2;
        while (runLength--)
          single[slot++] = ++previous;
      } else {
        const next = BASE91_INVERSE[input.charCodeAt(pos++)];
        previous += 89 + // eslint-disable-next-line unicorn/prefer-minimal-ternary -- branches read a different number of side-effecting input bytes
        (next < 90 ? next * base + BASE91_INVERSE[input.charCodeAt(pos++)] : BASE91_INVERSE[input.charCodeAt(pos++)] * 8281 + BASE91_INVERSE[input.charCodeAt(pos++)] * base + BASE91_INVERSE[input.charCodeAt(pos++)]);
        single[slot++] = previous;
      }
    }
  }
  decodeDelta(dict1AtomCount, 0);
  decodeDelta(dict2AtomCount, dictSize);
  const references = new Int32Array(ngramCount * 2);
  let poolSize = 0;
  let ngramIndex = 0;
  function readNgramReferences(count, startSlot) {
    for (let index = 0; index < count; index++) {
      const slot = startSlot + index;
      const a = readSlotCode();
      const b = readSlotCode();
      references[ngramIndex * 2] = a;
      references[ngramIndex * 2 + 1] = b;
      ngramIndex += 1;
      start[slot] = poolSize;
      const entryLength = (single[a] < 0 ? length[a] : 1) + (single[b] < 0 ? length[b] : 1);
      length[slot] = entryLength;
      poolSize += entryLength;
    }
  }
  readNgramReferences(ngramCount - dictSize + dict1AtomCount, dictSize + dict2AtomCount);
  readNgramReferences(dictSize - dict1AtomCount, dict1AtomCount);
  const pool = new Uint16Array(poolSize);
  let write = 0;
  for (let index = 0; index < ngramIndex; index++) {
    for (let half = 0; half < 2; half++) {
      const source = references[index * 2 + half];
      const value = single[source];
      if (value < 0) {
        let read = start[source];
        const readEnd = read + length[source];
        while (read < readEnd)
          pool[write++] = pool[read++];
      } else {
        pool[write++] = value;
      }
    }
  }
  const out = new Uint16Array(resultLength);
  let outIndex = 0;
  while (pos < inputLength) {
    let slot = BASE91_INVERSE[input.charCodeAt(pos++)];
    if (slot >= dictSize) {
      slot = slot * base - twoCharBias + BASE91_INVERSE[input.charCodeAt(pos++)];
    }
    const value = single[slot];
    if (value < 0) {
      let read = start[slot];
      const readEnd = read + length[slot];
      while (read < readEnd)
        out[outIndex++] = pool[read++];
    } else {
      out[outIndex++] = value;
    }
  }
  return out;
}

function determineBranch(decodeTree, current, nodeIndex, char) {
  const branchCount = (current & BinTrieFlags.BRANCH_LENGTH) >> 7;
  const jumpOffset = current & BinTrieFlags.JUMP_TABLE;
  if (jumpOffset) {
    if (branchCount === 0) {
      return char === jumpOffset ? nodeIndex : -1;
    }
    const slot = char - jumpOffset;
    if (slot >>> 0 >= branchCount)
      return -1;
    const stored = decodeTree[nodeIndex + slot];
    return stored === 0 ? -1 : nodeIndex + branchCount + stored - 1 & 65535;
  }
  if (branchCount === 0)
    return -1;
  const packedKeySlots = branchCount + 1 >> 1;
  const branchEnd = nodeIndex + packedKeySlots + branchCount;
  for (let index = 0; index < branchCount; index++) {
    const packed = decodeTree[nodeIndex + (index >> 1)];
    const key = packed >> ((index & 1) << 3) & 255;
    if (key === char) {
      const pointerIndex = nodeIndex + packedKeySlots + index;
      return branchEnd + decodeTree[pointerIndex] & 65535;

function parseNumericEntity(input, numberStart, inputLength) {
  let offset = numberStart + 1;
  let cp = 0;
  let digitStart = offset;
  if (offset < inputLength && (input.charCodeAt(offset) | TO_LOWER_BIT) === CharCodes.LOWER_X) {
    offset += 1;
    digitStart = offset;
    while (offset < inputLength) {
      const char = input.charCodeAt(offset);
      if (isNumber(char)) {
        cp = cp * 16 + (char - CharCodes.ZERO);
      } else if (isHexadecimalCharacter(char)) {
        cp = cp * 16 + ((char | TO_LOWER_BIT) - CharCodes.LOWER_A + 10);
      } else {
        break;
      }
      offset += 1;
    }
  } else {
    while (offset < inputLength) {
      const digit = input.charCodeAt(offset) - CharCodes.ZERO;
      if (digit >>> 0 > 9)
        break;
      cp = cp * 10 + digit;
      offset += 1;
    }
  }
  if (offset === digitStart)
    return 0;
  if (offset < inputLength && input.charCodeAt(offset) === CharCodes.SEMI) {
    offset += 1;

    longNumericConsumed = consumed;
    consumed = CONSUMED_OVERFLOW;
  }
  return consumed << CONSUMED_SHIFT | cp;
}

        if ((current & (BinTrieFlags.VALUE_LENGTH | BinTrieFlags.FLAG13)) === BinTrieFlags.FLAG13) {
          const runLength = (current & BinTrieFlags.BRANCH_LENGTH) >> 7;
          if (input.charCodeAt(index) !== (current & BinTrieFlags.JUMP_TABLE)) {
            break;
          }
          index += 1;
          const remaining = runLength - 1;
          let wordIndex = nodeIndex + 1;
          let charIndexInPacked = 0;
          for (; charIndexInPacked + 1 < remaining; charIndexInPacked += 2) {
            const packed = decodeTree[wordIndex];
            if (input.charCodeAt(index) !== (packed & 255))
              break trie;
            index += 1;
            if (input.charCodeAt(index) !== (packed >> 8 & 255))
              break trie;
            index += 1;
            wordIndex += 1;
          }
          if (charIndexInPacked < remaining) {
            if (input.charCodeAt(index) !== (decodeTree[wordIndex] & 255))
              break;
            index += 1;
          }
          nodeIndex += 1 + (runLength >> 1);
          current = decodeTree[nodeIndex];
          continue;
        }
        const valueLength = current >>> 14;
        const char = input.charCodeAt(index);
        if (valueLength !== 0) {
          if (char === CharCodes.SEMI) {
            consumed = index - entityStart + 1;
            value = valueLength === 1 ? String.fromCharCode(current & BinTrieFlags.VALUE_MASK) : readTrieValue(decodeTree, nodeIndex, valueLength);
            break;
          }
          if (!isStrict && (current & BinTrieFlags.FLAG13) === 0) {
            consumed = index - entityStart;
            bestNodeIndex = nodeIndex;
            bestValueLength = valueLength;
          }
          if (valueLength === 1)
            break;
        }
        const next = determineBranch(decodeTree, current, nodeIndex + (valueLength || 1), char);
        if (next < 0)
          break;
        nodeIndex = next;
        current = decodeTree[nodeIndex];
        index += 1;
      }
      if (value === "") {
        const finalVL = current >>> 14;
        if (finalVL !== 0 && !isStrict && (current & BinTrieFlags.FLAG13) === 0) {
          consumed = index - entityStart;
          bestNodeIndex = nodeIndex;
          bestValueLength = finalVL;
        }
        if (consumed > 0) {
          value = readTrieValue(decodeTree, bestNodeIndex, bestValueLength);
        }
      }
    } else {
      consumed = 0;
      value = "";
    }
    if (consumed === 0 || isAttribute && firstChar !== CharCodes.NUM && input.charCodeAt(entityStart + consumed - 1) !== CharCodes.SEMI && entityStart + consumed < inputLength && isEntityInAttributeInvalidEnd(input.charCodeAt(entityStart + consumed))) {
      offset = entityStart;
    } else {
      if (chunkStart < offset) {
        result += input.slice(chunkStart, offset);
      }
      result += value;
      offset = chunkStart = entityStart + consumed;
    }
    if (input.charCodeAt(offset) !== CharCodes.AMP) {
      offset = input.indexOf("&", offset);
    }
  } while (offset >= 0);
  return result + input.slice(chunkStart);
}

async function readBounded(response) {
  if (Number(response.headers.get("content-length")) > LIMITS2.bytes) {
    await response.body?.cancel();
    throw new InputError("Page exceeds the 2 MiB size limit.", 422);
  }
  if (!response.body) throw new InputError("The source returned an empty page.", 422);
  const reader = response.body.getReader(), chunks = [];
  let size = 0;
  try {
    while (true) {
      const { value, done } = await reader.read();
      if (done) break;
      size += value.byteLength;

        await reader.cancel();
        throw new InputError("Page exceeds the 2 MiB size limit.", 422);
      }

  }

}

}
async function tablesAPI(request, url) {
  if (request.method !== "POST") return new Response("Method not allowed", { status: 405, headers: { ...SECURITY, Allow: "POST" } });
  const origin = request.headers.get("origin");
  if (origin && origin !== url.origin) return json({ error: "Cross-origin extraction is not supported." }, 403);
  if (!/^application\/json(?:;|$)/i.test(request.headers.get("content-type") || "")) return json({ error: "Send a JSON request." }, 415);
  const checks = [];
  let phase = "Request validation";
  try {
    if (Number(request.headers.get("content-length")) > 4096) throw new InputError("Request is too large.", 413);
    const reader = request.body?.getReader();
    const chunks = [];
    let bytes = 0;
    if (reader) {
      try {
        while (true) {
          const { value, done } = await reader.read();
          if (done) break;
          bytes += value.byteLength;
          if (bytes > 4096) {
            await reader.cancel();
            throw new InputError("Request is too large.", 413);
          }
          chunks.push(value);
        }
      } finally {
        reader.releaseLock();
      }
    }
    const text = await new Blob(chunks).text();
    let body;
    try {
      body = JSON.parse(text);
    } catch {
      throw new InputError("Invalid JSON request.");
    }
    if (!body || typeof body.url !== "string") throw new InputError("Provide a public HTTPS URL.");
    checks.push({ step: "Request validation", status: "passed", message: "A bounded JSON request with a URL was received." });
    phase = "Public page request";
    const { source, page } = await fetchPublicPage(body.url, url.hostname);
    checks.push({ step: phase, status: "passed", message: "Public HTTPS and DNS checks passed for the request and redirects. HTML downloaded within the size and time limits." });
    phase = "Structure detection";
    const result = await analyzePage(page, source, checks);
    const diagnostics = { code: result.code, summary: result.message, next_action: result.next_action, checks, signals: result.signals };
    if (!result.tables.length) return json({ error: result.message, source, diagnostics }, 422);
    return json({ source, retrieved_at: (/* @__PURE__ */ new Date()).toISOString(), tables: result.tables, diagnostics });
  } catch (error) {
    const message = error.name === "TimeoutError" || error.name === "AbortError" ? "The source exceeded the 15-second request limit." : error.message || "Table extraction failed.";
    checks.push({ step: phase, status: "failed", message });
    return json({ error: message, diagnostics: { code: "extraction_stopped", summary: message, next_action: phase === "Request validation" ? "Correct the request and try again." : "Resolve the failed check before retrying. A later extraction strategy cannot bypass an access or request-limit failure.", checks } }, error.status || 502);
  }
}
