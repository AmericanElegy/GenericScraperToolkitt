  function highlight(paragraph, ranges) {
    const fragment = document.createDocumentFragment();
    let cursor = 0;
    const merged = [];
    for (const r of ranges.sort((a, b) => a.start - b.start)) {
      const last = merged.at(-1);
      if (last && r.start <= last.end) last.end = Math.max(last.end, r.end);
      else merged.push({ ...r });
    }
    for (const r of merged) {
      fragment.append(
        document.createTextNode(paragraph.slice(cursor, r.start)),
      );
      fragment.append(text("mark", paragraph.slice(r.start, r.end)));
      cursor = r.end;
    }
    fragment.append(document.createTextNode(paragraph.slice(cursor)));
    return fragment;
  }

  async function perform(task) {
    if (state.busy) return;
    state.busy = true;
    controls();

      await task();

      );
    } finally {
      state.busy = false;
      controls();
    }
  }
  function editCandidate(add = false) {
    const row = add ? null : selected();
    state.editingId = row?.id || null;
    $("correction-title").textContent = row ? "Correct " + row.id : "Add a new record";
    $("correction-note").value = "";
    const nodes = state.view.job.packet.schema.fields.map((f) => {
      const section = document.createElement("section");
      section.dataset.column = f.name;
      const field = row?.fields[f.name], first = field?.evidence[0];
      const valueLabel = text("label", f.name + " value");
      const value = document.createElement("input");
      value.dataset.part = "value"; value.required = f.required;
      value.type = f.type === "number" ? "number" : "text";
      if (f.type === "number") value.step = "any";
      value.value = field?.value ?? "";
      valueLabel.append(value);
      const passageLabel = text("label", "Supporting paragraph");
      const passage = document.createElement("select"); passage.dataset.part = "passage";
      for (const p of state.view.job.packet.passages) {
        const option = text("option", p.id + " · " + p.text.slice(0, 70));
        option.value = p.id; passage.append(option);
      }
      passage.value = first?.passage_id || state.view.job.packet.passages[0].id;
      passageLabel.append(passage);
      const quoteLabel = text("label", "Exact quotation");
      const quote = document.createElement("textarea"); quote.dataset.part = "quote";
      quote.required = f.required; quote.maxLength = 2048; quote.value = first?.quote || "";
      quoteLabel.append(quote);
      const occurrenceLabel = text("label", "Quotation occurrence (zero-based)");
      const occurrence = document.createElement("input"); occurrence.type = "number"; occurrence.min = "0";
      occurrence.step = "1"; occurrence.required = true; occurrence.dataset.part = "occurrence";
      occurrence.value = first?.occurrence ?? 0; occurrenceLabel.append(occurrence);
      section.append(valueLabel, passageLabel, quoteLabel, occurrenceLabel);
      return section;
    });
    $("correction-fields").replaceChildren(...nodes);
    controls();
  }
