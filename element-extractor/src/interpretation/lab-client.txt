  function highlight(paragraph, ranges) {
    const fragment = document.createDocumentFragment();
    let cursor = 0;
    const merged = [];
    for (const r of ranges.sort((a, b) => a.start - b.start)) {
      const last = merged.at(-1);
      if (last && r.start <= last.end) last.end = Math.max(last.end, r.end);
      else merged.push({ ...r });
    }
    for (const r of merged) {
      fragment.append(
        document.createTextNode(paragraph.slice(cursor, r.start)),
      );
      fragment.append(text("mark", paragraph.slice(r.start, r.end)));
      cursor = r.end;
    }
    fragment.append(document.createTextNode(paragraph.slice(cursor)));
    return fragment;
  }

      state.savedReviews=state.savedReviews.filter(s=>s.session!==key&&!result.conflicts.includes(s.session));

      state.feedback=state.feedback.slice(0,20);storageInventory(result);feedbackSummary();

  async function autoSaveFeedback(job){
    state.unsavedFeedback.set(job.packet.hash,job);
    try{await storageReady;if(!state.storage?.enabled){storageMessage("Feedback not saved to disk. Connect your Drive folder or keep a portable backup before closing.");return;}await savePendingFeedback();}
    catch(error){storageMessage("Feedback not saved: "+error.message+" Review decisions are still in this tab. Reconnect Drive and retry, or save the review session before closing.");}
  }
  $("connect-storage").onclick=()=>perform(async()=>{
    await storageReady;
    state.storage=await storageRequest({action:"configure",drive_root:$("drive-root").value.trim()});
    await loadStorage();
    if(state.view?.job.events.length||state.view?.job.revisions.length)state.unsavedFeedback.set(state.view.job.packet.hash,state.view.job);
    await savePendingFeedback();
  });
  $("reload-storage").onclick=()=>perform(async()=>{await storageReady;await loadStorage();await savePendingFeedback();});
  window.addEventListener("beforeunload",event=>{if(state.unsavedFeedback.size){event.preventDefault();event.returnValue="";}});
  const stableFeedback=value=>Array.isArray(value)?"["+value.map(stableFeedback).join(",")+"]":value&&typeof value==="object"?"{"+Object.keys(value).sort().map(k=>JSON.stringify(k)+":"+stableFeedback(value[k])).join(",")+"}":JSON.stringify(value);
  $("open-feedback-review").onclick=()=>perform(async()=>{
    await storageReady;const item=feedbackChoices().find(c=>c.id===$("saved-feedback").value);if(!item)return;
    const response=await request("/api/interpretation/action",{action:"restore",session:{format:"toolkit-review-session-v1",job:item.job}});
    state.view=await response.json();state.rowId=null;state.field=null;clearRetrieved();clearDecision();render();editCandidate();
    $("source-text").value=state.view.job.document?.source.text||state.view.job.packet.source.text;
    $("source-url").value=state.view.job.packet.source.url||"";$("reviewer").value="";$("batch-reviewer").value="";
    $("text-mode").value=state.view.job.parser_version==="name-number-v1"?"pair":"auto";
    if($("text-mode").value==="pair"){$("name-column").value=state.view.job.packet.schema.fields[0].name;$("number-column").value=state.view.job.packet.schema.fields[1].name;}
    controls();say("Saved feedback review reopened and independently revalidated. No source fetch or AI call was made.");
  });
  function feedbackChoices(){
    const saved=state.savedReviews.map(s=>({id:s.id,job:s.example?.job||s.job,expectations:s.example?.expectations}));
    return [...saved,...state.feedback.filter(c=>!saved.some(s=>s.job.packet.hash===c.job.packet.hash))];
  }
  function feedbackSummary(){
    const selectedId=$("saved-feedback").value;
    $("saved-feedback").replaceChildren(...feedbackChoices().map(c=>{const label=c.job.retrieval?.label||c.job.packet.source.url||c.job.parser_version||"Scripted example";const option=text("option",label+(c.expectations?" · "+c.expectations.approved.length+" approved · "+c.expectations.rejected.length+" rejected":" · pending review"));option.value=c.id;return option;}));
    if(feedbackChoices().some(c=>c.id===selectedId))$("saved-feedback").value=selectedId;
    $("feedback-summary").textContent=state.feedback.length+" reviewed examples collected. "+(state.storage?.enabled?"Reviewed changes save automatically to the connected folder. Downloads are optional backups.":"Download before closing; Drive saving is not connected.")+" New source pages do not clear this collection.";
  }
  const feedbackText=items=>JSON.stringify({format:"toolkit-feedback-collection-v1",cases:items},null,2)+"\n";
  function addFeedback(items){
    const merged=[...state.feedback];
    for(const item of items)if(!merged.some(c=>c.id===item.id))merged.push(item);
    if(merged.length>20||new Blob([feedbackText(merged)]).size>5000000)throw Error("A collection supports up to 20 cases or 5 MB. Download this collection and start another tab for more.");
    state.feedback=merged;feedbackSummary();controls();
  }
  $("collect-feedback").onclick=()=>perform(async()=>{
    await storageReady;
    const response=await request("/api/interpretation/action",{action:"feedback",job:state.view.job});
    addFeedback([await response.json()]);await autoSaveFeedback(state.view.job);say("Reviewed example collected. See Feedback autosave for disk-save status. No extraction rules were changed.");
  });
  $("download-feedback").onclick=()=>{
    saveBlob(new Blob([feedbackText(state.feedback)],{type:"application/json"}),"toolkit-feedback-collection.json");
    say("Feedback collection downloaded. Open it here later to add more cases, or run the offline feedback check against a future update.");
  };
  $("import-feedback").onclick=()=>perform(async()=>{
    await storageReady;
    const file=$("feedback-file").files[0];if(!file||file.size>5000000)throw Error("Choose a feedback collection JSON file up to 5 MB.");
    const collection=JSON.parse(await file.text());
    if(collection.format!=="toolkit-feedback-collection-v1"||!Array.isArray(collection.cases)||!collection.cases.length||collection.cases.length>20)throw Error("Choose a collection containing 1–20 reviewed examples.");
    const verified=[];
    for(const item of collection.cases){
      const response=await request("/api/interpretation/action",{action:"feedback",job:item.job});const actual=await response.json();
      if(item.format!==actual.format||item.id!==actual.id||stableFeedback(item)!==stableFeedback(actual))throw Error("A feedback case differs from its validated review history. Your current collection was kept.");
      verified.push(actual);
    }
    addFeedback(verified);for(const item of verified)await autoSaveFeedback(item.job);say("Feedback collection opened and review histories revalidated. Current extraction is unchanged.");
  });
  $("save-session").onclick = () => perform(async () => {
    const response = await request("/api/interpretation/action", { action: "save", job: state.view.job });
    saveBlob(await response.blob(), "toolkit-review-session.json");
    say("Saved the current review session. Open this file on either computer to continue. Other sections have separate reviews.");
  });
  $("restore-form").onsubmit = event => {
    event.preventDefault();
    perform(async () => {
      const file = $("session-file").files[0];
      if (!file || file.size > 1000001) throw Error("Choose a saved session JSON file up to 1,000,001 bytes.");
      let session;
      try { session = JSON.parse(await file.text()); } catch { throw Error("This file is not valid session JSON. Your current review was kept."); }
      const response = await request("/api/interpretation/action", { action: "restore", session });
      state.view = await response.json(); state.rowId = null; state.field = null;
      clearRetrieved();clearDecision(); render(); editCandidate();
      $("source-text").value = state.view.job.document?.source.text || state.view.job.packet.source.text;
      $("source-url").value = state.view.job.packet.source.url || "";
      $("reviewer").value = "";
      $("batch-reviewer").value = "";
      $("text-mode").value = state.view.job.parser_version === "name-number-v1" ? "pair" : "auto";
      if ($("text-mode").value === "pair") {
        $("name-column").value = state.view.job.packet.schema.fields[0].name;
        $("number-column").value = state.view.job.packet.schema.fields[1].name;
      }
      controls();
      say("Saved session opened and independently revalidated. Imported review names are self-declared; compare the retained source before relying on them.");
    });
  };
  async function download(format, diagnostic) {
    const response = await request("/api/interpretation/action", {
      action: "export",
      job: state.view.job,
      format,
      diagnostic,
    });
    saveBlob(await response.blob(), (diagnostic ? "interpretation-diagnostic" : "interpretation-approved") + "." + format);
    say(
      "Downloaded " +
        response.headers.get("X-Exported-Records") +
        " " +
        (diagnostic ? "diagnostic" : "approved") +
        " records.",
    );
  }
  $("approved-csv").onclick = () => perform(() => download("csv", false));
  $("approved-json").onclick = () => perform(() => download("json", false));
  $("diagnostic-json").onclick = () => perform(() => download("json", true));
  controls();
  if(workspaceToken&&window.opener)window.opener.postMessage({channel:'toolkit-workspace-v1',token:workspaceToken,action:'ready'},location.origin);
  storageReady=loadStorage().catch(error=>storageMessage("Feedback storage could not be loaded: "+error.message+" Keep a portable backup and retry."));
})();
