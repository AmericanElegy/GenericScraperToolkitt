  async function request(path, body) {
    const response = await fetch(path, {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify(body),
    });
    if (!response.ok) {
      let error;
      try {
        error = await response.json();
      } catch {}
      throw Error(
        error?.error ||
          "The lab request failed. Your current session was kept.",
      );
    }
    return response;
  }

  function highlight(paragraph, ranges) {
    const fragment = document.createDocumentFragment();
    let cursor = 0;
    const merged = [];
    for (const r of ranges.sort((a, b) => a.start - b.start)) {
      const last = merged.at(-1);
      if (last && r.start <= last.end) last.end = Math.max(last.end, r.end);
      else merged.push({ ...r });
    }
    for (const r of merged) {
      fragment.append(
        document.createTextNode(paragraph.slice(cursor, r.start)),
      );
      fragment.append(text("mark", paragraph.slice(r.start, r.end)));
      cursor = r.end;
    }
    fragment.append(document.createTextNode(paragraph.slice(cursor)));
    return fragment;
  }

          .map((e) => ({ start: e.passage_start, end: e.passage_end }));

      }),

  }
  function displayValue(row, field) {
    const value = row.fields[field.name].value;
    if (value === null || value === undefined) return "—";
    if (typeof value === "number" && /^(?:hourly|annual)_wage(?:_.*)?_dollars$/.test(field.name))
      return "$" + new Intl.NumberFormat("en-US", { minimumFractionDigits: 2, maximumFractionDigits: 20 }).format(value);
    if (typeof value === "number" && /^employment_percent_/.test(field.name)) return value + "%";
    if (typeof value === "number" && field.number_format === "en-US-grouped") return new Intl.NumberFormat("en-US", { maximumFractionDigits: 20 }).format(value);
    return value;
  }

            state.rowId = row.id;

          ),

        return tr;
      }),

      columnLabel(state.field) + " · record " + row.id.replace(/^row-/, "");

      );

        (r) => r.id === row.id,

        evidence.push(
          text("blockquote", e.quote),
          text("p", "Proposed quotation · not validated", "hint"),
        );

    $("evidence").replaceChildren(...evidence);
    $("findings-details").open = blocking(row);

    for (const finding of [
      ...state.view.validation.issues,
      ...row.issues,
      ...Object.values(row.fields).flatMap((f) => f.issues),

      list.append(text("li", finding.message));

      text("strong", row.review_status === "approved" ? "Original automated findings · human association review completed" : "System findings — retained after review"),
      list,
    );

      "Review " +

      " · " +
      (labels[row.review_status] || row.review_status);

  async function perform(task) {
    if (state.busy) return;
    state.busy = true;
    controls();
    try {
      await task();
    } catch (error) {
      say(
        error.message || "The request failed. Your current session was kept.",
      );
    } finally {
      state.busy = false;
      controls();
    }
  }
  function editCandidate(add = false) {
    const row = add ? null : selected();
    state.editingId = row?.id || null;
    $("correction-title").textContent = row ? "Correct " + row.id : "Add a new record";
    $("correction-note").value = "";
    const nodes = state.view.job.packet.schema.fields.map((f) => {
      const section = document.createElement("section");
      section.dataset.column = f.name;
      const field = row?.fields[f.name], first = field?.evidence[0];
      const valueLabel = text("label", f.name + " value");
      const value = document.createElement("input");
      value.dataset.part = "value"; value.required = f.required;
      value.type = f.type === "number" ? "number" : "text";
      if (f.type === "number") value.step = "any";
      value.value = field?.value ?? "";
      valueLabel.append(value);
      const passageLabel = text("label", "Supporting paragraph");
      const passage = document.createElement("select"); passage.dataset.part = "passage";
      for (const p of state.view.job.packet.passages) {
        const option = text("option", p.id + " · " + p.text.slice(0, 70));
        option.value = p.id; passage.append(option);
      }
      passage.value = first?.passage_id || state.view.job.packet.passages[0].id;
      passageLabel.append(passage);
      const quoteLabel = text("label", "Exact quotation");
      const quote = document.createElement("textarea"); quote.dataset.part = "quote";
      quote.required = f.required; quote.maxLength = 2048; quote.value = first?.quote || "";
      quoteLabel.append(quote);
      const occurrenceLabel = text("label", "Quotation occurrence (zero-based)");
      const occurrence = document.createElement("input"); occurrence.type = "number"; occurrence.min = "0";
      occurrence.step = "1"; occurrence.required = true; occurrence.dataset.part = "occurrence";
      occurrence.value = first?.occurrence ?? 0; occurrenceLabel.append(occurrence);
      section.append(valueLabel, passageLabel, quoteLabel, occurrenceLabel);
      return section;
    });
    $("correction-fields").replaceChildren(...nodes);
    controls();
  }

        "Approval requires clean checks and explicit association confirmation.",

    }

      const response = await request("/api/interpretation/action", {
        action: "review",
        job: state.view.job,
        decision: {
          record_id: row.id,
          decision,
          reviewer: $("reviewer").value.trim(),
          note: $("note").value.trim(),
          confirmed_association: $("confirm").checked,
        },
      });
      state.view = await response.json();
      clearDecision();
      render();
      say(
        "Appended " +
          labels[decision].toLowerCase() +
          " decision for " +
          row.id +
          ". Earlier events were kept.",
      );
    });
  };
  $("revalidate").onclick = () =>
    perform(async () => {
      const response = await request("/api/interpretation/action", {
        action: "revalidate",
        job: state.view.job,
      });
      state.view = await response.json();
      render();
      say(
        "Cached scripted response revalidated. No interpreter or AI call was made.",
      );
    });
  function saveBlob(blob, filename) {
    const url = URL.createObjectURL(blob), anchor = document.createElement("a");
    anchor.href = url; anchor.download = filename; anchor.click();
    setTimeout(() => URL.revokeObjectURL(url), 1000);
  }
  $("save-session").onclick = () => perform(async () => {
    const response = await request("/api/interpretation/action", { action: "save", job: state.view.job });
    saveBlob(await response.blob(), "toolkit-review-session.json");
    say("Saved the current review session. Open this file on either computer to continue. Other sections have separate reviews.");
  });
  $("restore-form").onsubmit = event => {
    event.preventDefault();
    perform(async () => {
      const file = $("session-file").files[0];
      if (!file || file.size > 1000001) throw Error("Choose a saved session JSON file up to 1,000,001 bytes.");
      let session;
      try { session = JSON.parse(await file.text()); } catch { throw Error("This file is not valid session JSON. Your current review was kept."); }
      const response = await request("/api/interpretation/action", { action: "restore", session });
      state.view = await response.json(); state.rowId = null; state.field = null;
      clearDecision(); render(); editCandidate();
      $("source-text").value = state.view.job.document?.source.text || state.view.job.packet.source.text;
      $("source-url").value = state.view.job.packet.source.url || "";
      $("reviewer").value = "";
      $("batch-reviewer").value = "";
      $("text-mode").value = state.view.job.parser_version === "name-number-v1" ? "pair" : "auto";
      if ($("text-mode").value === "pair") {
        $("name-column").value = state.view.job.packet.schema.fields[0].name;
        $("number-column").value = state.view.job.packet.schema.fields[1].name;
      }
      controls();
      say("Saved session opened and independently revalidated. Imported review names are self-declared; compare the retained source before relying on them.");
    });
  };
  async function download(format, diagnostic) {
    const response = await request("/api/interpretation/action", {
      action: "export",
      job: state.view.job,
      format,
      diagnostic,
    });
    saveBlob(await response.blob(), (diagnostic ? "interpretation-diagnostic" : "interpretation-approved") + "." + format);
    say(
      "Downloaded " +
        response.headers.get("X-Exported-Records") +
        " " +
        (diagnostic ? "diagnostic" : "approved") +
        " records.",
    );
  }
  $("approved-csv").onclick = () => perform(() => download("csv", false));
  $("approved-json").onclick = () => perform(() => download("json", false));
  $("diagnostic-json").onclick = () => perform(() => download("json", true));
  controls();
})();
